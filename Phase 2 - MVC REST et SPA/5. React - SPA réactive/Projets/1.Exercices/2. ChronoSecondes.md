*App.jsx*
```js
import { useState } from "react"
export function App(){
    const [compteur,setCompteur] = useState(0);
    function handleClick()
    {        
        setCompteur(compteur+1);    
    }    

    return (
        <div>
            <p>{compteur}</p> 
            <button onClick={handleClick}>+</button>        
        </div>    
    )   
}

```
